# .github/workflows/ci.yml
# CI workflow with caching for pyenv and Python installations
# Inspired by grug-lang/grug-ir's build.yml caching strategy

name: CI

on:
  push:
    branches: [ main, master ]
  pull_request:
    branches: [ main, master ]

jobs:
  build:
    runs-on: ubuntu-latest
    
    steps:
      # Step 1: Checkout repository
      - name: Checkout code
        uses: actions/checkout@v3

      # Step 2: Cache pyenv and Python installations
      # This reduces 'Install pyenv and build dependencies' from 6s to ~1-2s
      # and 'Install Python' from 1m51s to ~2-3s
      - name: Cache pyenv and Python
        id: cache-pyenv
        uses: actions/cache@v3
        with:
          # Cache the entire pyenv directory including Python installations
          path: |
            ~/.pyenv
            ~/.pyenv_cache
          # Generate cache key based on:
          # - OS (ubuntu-latest)
          # - Python version (3.7.17)
          # - Date (weekly rotation to prevent cache bloat)
          key: ${{ runner.os }}-pyenv-${{ hashFiles('**/.python-version') }}-${{ github.run_id }}
          # Fallback to latest cache if exact match not found
          restore-keys: |
            ${{ runner.os }}-pyenv-${{ hashFiles('**/.python-version') }}-
            ${{ runner.os }}-pyenv-

      # Step 3: Install pyenv and build dependencies (only if cache miss)
      - name: Install pyenv and build dependencies
        if: steps.cache-pyenv.outputs.cache-hit != 'true'
        run: |
          # Install pyenv dependencies
          sudo apt-get update
          sudo apt-get install -y \
            make \
            build-essential \
            libssl-dev \
            zlib1g-dev \
            libbz2-dev \
            libreadline-dev \
            libsqlite3-dev \
            wget \
            curl \
            llvm \
            libncurses5-dev \
            xz-utils \
            tk-dev \
            libxml2-dev \
            libxmlsec1-dev \
            libffi-dev \
            liblzma-dev

          # Install pyenv
          curl https://pyenv.run | bash
          
          # Configure pyenv
          echo 'export PYENV_ROOT="$HOME/.pyenv"' >> ~/.bashrc
          echo 'command -v pyenv >/dev/null || export PATH="$PYENV_ROOT/bin:$PATH"' >> ~/.bashrc
          echo 'eval "$(pyenv init -)"' >> ~/.bashrc
          
          # Source pyenv immediately for this step
          export PYENV_ROOT="$HOME/.pyenv"
          export PATH="$PYENV_ROOT/bin:$PATH"
          eval "$(pyenv init -)"

      # Step 4: Install Python 3.7.17 (only if cache miss)
      - name: Install Python
        if: steps.cache-pyenv.outputs.cache-hit != 'true'
        run: |
          # Source pyenv
          export PYENV_ROOT="$HOME/.pyenv"
          export PATH="$PYENV_ROOT/bin:$PATH"
          eval "$(pyenv init -)"
          
          # Install Python 3.7.17
          pyenv install 3.7.17
          pyenv global 3.7.17
          
          # Verify installation
          python --version
          which python

      # Step 5: Set up Python environment (always runs)
      - name: Setup Python environment
        run: |
          # Source pyenv
          export PYENV_ROOT="$HOME/.pyenv"
          export PATH="$PYENV_ROOT/bin:$PATH"
          eval "$(pyenv init -)"
          
          # Ensure correct Python version is active
          pyenv global 3.7.17
          
          # Verify Python version
          python --version
          echo "Python location: $(which python)"

      # Step 6: Install project dependencies
      - name: Install dependencies
        run: |
          # Source pyenv
          export PYENV_ROOT="$HOME/.pyenv"
          export PATH="$PYENV_ROOT/bin:$PATH"
          eval "$(pyenv init -)"
          
          # Upgrade pip
          pip install --upgrade pip
          
          # Install project dependencies (adjust as needed)
          if [ -f requirements.txt ]; then
            pip install -r requirements.txt
          fi
          
          # Install build tools
          pip install wheel setuptools

      # Step 7: Run tests
      - name: Run tests
        run: |
          # Source pyenv
          export PYENV_ROOT="$HOME/.pyenv"
          export PATH="$PYENV_ROOT/bin:$PATH"
          eval "$(pyenv init -)"
          
          # Run tests (adjust test command as needed)
          if [ -f setup.py ] || [ -f pyproject.toml ]; then
            python -m pytest tests/ -v
          else
            echo "No test configuration found"
          fi

      # Step 8: Build project (optional)
      - name: Build project
        if: github.event_name == 'push' && (github.ref == 'refs/heads/main' || github.ref == 'refs/heads/master')
        run: |
          # Source pyenv
          export PYENV_ROOT="$HOME/.pyenv"
          export PATH="$PYENV_ROOT/bin:$PATH"
          eval "$(pyenv init -)"
          
          # Build package (adjust as needed)
          if [ -f setup.py ]; then
            python setup.py sdist bdist_wheel
          elif [ -f pyproject.toml ]; then
            pip install build
            python -m build
          fi

      # Step 9: Upload artifacts (optional)
      - name: Upload build artifacts
        if: github.event_name == 'push' && (github.ref == 'refs/heads/main' || github.ref == 'refs/heads/master')
        uses: actions/upload-artifact@v3
        with:
          name: dist
          path: dist/

# Test workflow to verify caching behavior
# .github/workflows/test-cache.yml
name: Test Cache

on:
  workflow_dispatch:  # Manual trigger for testing

jobs:
  test-cache:
    runs-on: ubuntu-latest
    
    steps:
      - name: Checkout code
        uses: actions/checkout@v3

      - name: Test cache restoration
        id: test-cache
        uses: actions/cache@v3
        with:
          path: |
            ~/.pyenv
            ~/.pyenv_cache
          key: ${{ runner.os }}-pyenv-test-${{ github.run_id }}
          restore-keys: |
            ${{ runner.os }}-pyenv-test-
            ${{ runner.os }}-pyenv-

      - name: Verify Python installation from cache
        run: |
          if [ -d "$HOME/.pyenv" ]; then
            echo "pyenv directory exists"
            export PYENV_ROOT="$HOME/.pyenv"
            export PATH="$PYENV_ROOT/bin:$PATH"
            eval "$(pyenv init -)"
            
            # Check if Python 3.7.17 is installed
            if pyenv versions | grep -q "3.7.17"; then
              echo "Python 3.7.17 is installed from cache"
              pyenv global 3.7.17
              python --version
            else
              echo "Python 3.7.17 not found in cache"
              exit 1
            fi
          else
            echo "No pyenv cache found"
            exit 1
          fi

      - name: Measure cache restoration time
        run: |
          echo "Cache restoration completed"
          echo "Expected time: < 5 seconds for cached steps"
          echo "Actual time will be shown in workflow run logs"
